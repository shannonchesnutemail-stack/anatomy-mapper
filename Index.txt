  function mapUrl(q){return "https://www.google.com/maps/search/?api=1&query="+encodeURIComponent(q);}

  // Chain every stop (Day 1 + Day 2) into one Google Maps route, starting & ending in Lincoln
  function routeUrl(trip){
    const stops=[...trip.d1,...trip.d2].map(r=>r[2]); // map queries in order
    const waypoints=["Lincoln, NE", ...stops, "Lincoln, NE"];
    const origin=encodeURIComponent(waypoints[0]);
    const destination=encodeURIComponent(waypoints[waypoints.length-1]);
    const mid=waypoints.slice(1,-1).map(encodeURIComponent).join('|');
    return "https://www.google.com/maps/dir/?api=1&origin="+origin+"&destination="+destination+"&waypoints="+mid+"&travelmode=driving";
  }

  function tlRows(arr){
    return arr.map(([t,desc,q])=>`<li><span class="t">${t}</span><span>${desc} <a class="maplink" href="${mapUrl(q)}" target="_blank" rel="noopener">Map</a></span></li>`).join('');
  }
  function costRows(arr,total){
    return arr.map(([k,v])=>`<tr><td>${k}</td><td>${v}</td></tr>`).join('')+`<tr class="total"><td>Estimated Total</td><td>$${total}</td></tr>`;
  }

  function buildShareUrl(trip){
    const base=location.href.split('#')[0].split('?')[0];
    return base+'#trip='+encodeURIComponent(trip.id);
  }
  async function shareTrip(trip){
    const url=buildShareUrl(trip);
    const text=`Shannon & Brandi's getaway: ${trip.name.replace(/&amp;/g,'&')} (est. $${trip.cost} for two). View the full itinerary:`;
    if(navigator.share){
      try{await navigator.share({title:"Our Getaway Itinerary",text,url});return;}catch(e){}
    }
    try{
      await navigator.clipboard.writeText(url);
      showToast('Link copied to clipboard ✓');
    }catch(e){
      prompt('Copy this link to share:',url);
    }
  }

  function loadFromHash(){
    const m=location.hash.match(/trip=([^&]+)/);
    if(!m)return;
    const id=decodeURIComponent(m[1]);
    const trip=TRIPS.find(t=>t.id===id);
    if(!trip)return;
    current=trip;
    pickName.innerHTML=trip.name;pickCost.textContent='$'+trip.cost;bar.classList.add('show');
    document.querySelectorAll('.card').forEach(c=>{
      if(c.dataset.id===id){const b=c.querySelector('.choose');b.classList.add('picked');b.textContent='Selected';}
    });
    const sb=document.getElementById('sharedBanner');
    sb.innerHTML=`📩 Shared with you: <b>${trip.name}</b> — here's the full itinerary, ready to take on the road.`;
    sb.classList.add('show');
    setTimeout(()=>openTicket(trip),400);
